post with always
=================
Run the steps in the post section regardless of the completion status of the Pipeline’s or stage’s run.


pipeline {
    agent any
    stages {
        stage('Example1') {
            steps {
                echo 'Hello World'
            }
        }
         stage('Example2') {
            steps {
                sh 'touch file1'
            }
        }
         stage('Example3') {
            steps {
               sh 'touch file2'
            }
        }
    }
    post { 
        always { 
            echo 'I will always say Hello again!'
        }
    }
}
