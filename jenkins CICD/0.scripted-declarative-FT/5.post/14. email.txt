Email notifications through Jenkins:-
============================
1. Open the dashboard of Jenkins
2. Go to the job for which we want email notifications
3. Click on configure
4. Go to post build action

5. Click on post build actions
6. Click on email notification

7. Enter email ID's of team members separated by space

8. Apply--> same
9. Go to the dashboard of Jenkins
10. Click on manage Jenkins
11. Click on configure system
12. Go to email notification
	SMTP server 	smtp.gmail.com
	Default user email
13. Click on advanced
	Click use SMTP authentication
	Enter your Gmail username and password
14. Click on use SSL
15. SMTP port 465
16. Check on test configuration by test email and enter your gmail ID
17. Click on test configuration

pipeline {
    agent any

    stages {
        stage('Build') {
            steps {
                // your build steps
            }
        }
    }

    post {
        failure {
            script {
                def buildNumber = env.BUILD_NUMBER
                def jobName = env.JOB_NAME
                def buildUrl = env.BUILD_URL
                def timestamp = new Date().format("yyyy-MM-dd HH:mm:ss")

                // Archive logs as attachment
                archiveArtifacts artifacts: '**/target/*.log, **/logs/*.log', 
                                 allowEmptyArchive: true

                // Email notification with attachment
                emailext(
                    subject: "❌ BUILD FAILED: ${jobName} - Build #${buildNumber}",
                    body: """
                        <html>
                        <body>
                            <h2 style="color:red;">Build Failed!</h2>
                            <table>
                                <tr><td><b>Job Name:</b></td><td>${jobName}</td></tr>
                                <tr><td><b>Build Number:</b></td><td>#${buildNumber}</td></tr>
                                <tr><td><b>Build URL:</b></td><td><a href="${buildUrl}">${buildUrl}</a></td></tr>
                                <tr><td><b>Timestamp:</b></td><td>${timestamp}</td></tr>
                                <tr><td><b>Status:</b></td><td style="color:red;">FAILURE</td></tr>
                            </table>
                            <p>Please check the attached console log for more details.</p>
                        </body>
                        </html>
                    """,
                    mimeType: 'text/html',
                    to: 'team@example.com',
                    attachLog: true,                          // attaches console log
                    attachmentsPattern: '**/target/*.log'     // attaches additional logs
                )
            }
        }

        always {
            script {
                // Always archive test results if present
                junit allowEmptyResults: true, testResults: '**/target/surefire-reports/*.xml'
            }
        }

        success {
            echo "✅ Build #${env.BUILD_NUMBER} passed successfully."
        }

        unstable {
            emailext(
                subject: "⚠️ BUILD UNSTABLE: ${env.JOB_NAME} - Build #${env.BUILD_NUMBER}",
                body: """
                    Build #${env.BUILD_NUMBER} is unstable.
                    Check details at: ${env.BUILD_URL}
                """,
                to: 'team@example.com',
                attachLog: true
            )
        }
    }
}
