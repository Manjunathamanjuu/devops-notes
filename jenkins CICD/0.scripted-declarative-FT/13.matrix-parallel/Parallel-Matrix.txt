pipeline {
    parameters {
        choice(name: 'PLATFORM_FILTER', choices: ['all', 'redhat', 'ubuntu'], description: 'Run on specific platform')
    }
    agent none
    stages {
        stage('BuildAndTest') {
            matrix {
                agent {
                    label "${PLATFORM}"
                }
                when { anyOf {
                    expression { params.PLATFORM_FILTER == 'all' }
                    expression { params.PLATFORM_FILTER == env.PLATFORM }
                } }
                axes {
                    axis {
                        name 'PLATFORM'
                        values 'ubuntu', 'redhat'
                    }
                    axis {
                        name 'BROWSER'
                        values 'firefox', 'chrome', 'ie'
                    }
                }
                excludes {
                    exclude {
                        axis {
                            name 'PLATFORM'
                            values 'redhat'
                        }
                        axis {
                            name 'BROWSER'
                            values 'firefox'
                        }
                    }
                    exclude {
                        axis {
                            name 'PLATFORM'
                            notValues 'redhat'
                        }
                        axis {
                            name 'BROWSER'
                            values 'ie'
                        }
                    }
                }
                stages {
                    stage('Build') {
                        steps {
                            echo "Do Build for ${PLATFORM} - ${BROWSER}"
                        }
                    }
                    stage('Test') {
                        steps {
                            echo "Do Test for ${PLATFORM} - ${BROWSER}"
                        }
                    }
                }
            }
        }
    }
}
